FROM singer;
SELECT COUNT(*) FROM singer;
SELECT Name, Country, Age
FROM singer
ORDER BY Age DESC;
SELECT singer.Name, singer.Country, singer.Age
FROM singer
ORDER BY singer.Age DESC;
SELECT AVG(a.Age), MIN(a.Age) AS min_age, MAX(a.Age) AS max_age
FROM singer AS a
WHERE a.Country = 'France';
SELECT AVG(Age), MIN(Age), MAX(Age)
FROM singer
WHERE Country = 'France';
WITH MinAge AS (
    SELECT MIN(Age) AS MinAge
    FROM singer
),
YoungestSinger AS (
    SELECT Singer_ID, Name, Country, Age
    FROM singer
    JOIN MinAge ON singer.Age = MinAge.MinAge
),
ConcertDetails AS (
    SELECT concert.concert_Name, concert.Year, song_Name, song_release_year
    FROM concert
    JOIN singer_in_concert ON concert.concert_ID = singer_in_concert.concert_ID
    JOIN YoungestSinger ON singer_in_concert.Singer_ID = YoungestSinger.Singer_ID
)
SELECT Name, song_release_year FROM ConcertDetails;
WITH YoungestSinger AS (
  SELECT Singer_ID, MIN(Age) AS MinAge
  FROM singer
),
YoungestSingerConcerts AS (
  SELECT concert_ID
  FROM singer_in_concert
  JOIN singer ON singer_in_concert.Singer_ID = singer.Singer_ID
  JOIN YoungestSinger ON singer.Age = YoungestSinger.MinAge
)
SELECT Name, Song_release_year
FROM singer
JOIN YoungestSingerConcerts ON singer.Singer_ID = YoungestSingerConcerts.Singer_ID
WHERE Is_male = 1; -- Assuming the youngest singer is male (you can remove this condition if you want all singers)
SELECT DISTINCT singer.Country
FROM singer
JOIN singer_in_concert ON singer.Singer_ID = singer_in_concert.Singer_ID
WHERE singer.Age > 20;
SELECT Country
FROM singer

    SELECT Singer_ID
    FROM singer_in_concert
    JOIN concert ON concert.concert_ID = singer_in_concert.concert_ID
    JOIN stadium ON concert.Stadium_ID = stadium.Stadium_ID
